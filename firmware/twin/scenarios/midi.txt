# MIDI control on channel 16 (MidiControl.h): notes as keys, CCs, NRPN, a program change, a morph
input sine 220 0.3
at 6000
# a note holds the filter, CC 86 closes it, the note lets go
midi 9F 37 64
midi BF 56 10
wait 1500
# a note holds the filter: its knobs on the panel, CC 86 closing knob 1
expect led pth 1 3737fd
expect led pth 2 798fc6
midi 8F 37 00
wait 500
# CC 24 latches the filter, NRPN 86 sweeps its cutoff in 14 bits, CC 24 unlatches it
midi BF 18 7F
midi BF 63 00 BF 62 56 BF 06 20 BF 26 00
wait 1000
midi BF 06 60 BF 26 40
wait 1000
midi BF 18 00
wait 500
# the delay latched, its level and division set outright; CC 14 turns knob 1 (the freezer's)
midi BF 1E 7F BF 71 60 BF 6E 40
wait 1500
midi BF 0E 03
wait 500
# the blank scene by program change: every effect off
midi CF 00
wait 1500
# program change 0: the blank scene, its key bright
expect led smt 0 fcfcfc
# notes 45-47: LOOP records, LOOP again plays
midi 9F 2F 64
wait 60
# note 47: LOOP records
expect led pth 8 fd0000
midi 8F 2F 00
wait 2000
midi 9F 2F 64
wait 60
# and plays
expect led pth 6 79f200
midi 8F 2F 00
wait 1500
# morph to the blank scene over 2 bars
midi BF 1C 7F BF 3D 02 BF 3E 00
wait 5000
expect on
